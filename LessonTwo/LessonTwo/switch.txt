var son = Convert.ToInt32(Console.ReadLine());

if (son == 1)
{
    Console.WriteLine("Bir");
}
else if (son == 2)
{
    Console.WriteLine("Ikki");
}
else if(son == 3)
{
    Console.WriteLine("Uch");
}
else
{
    Console.WriteLine("Son");
}

switch (son) 
{ 
    case 1: Console.WriteLine("bir");break;
    case 2: Console.WriteLine("ikki");break;
    case 3: Console.WriteLine("uch");break;
    default: Console.WriteLine("son");break;
}
